    2. Add `togglePackToGo: (tableId: string, lineId: string) => void` to the OrderStore interface, then implement it in the create() body using the exact same immutable round-map pattern as voidItem:

    ```typescript
    togglePackToGo: (tableId, lineId) =>
      set((state) => {
        const existing = state.orders[tableId]
        if (!existing) return state
        const newRounds = existing.rounds.map((round) => ({
          ...round,
          items: round.items.map((item) =>
            item.lineId === lineId
              ? { ...item, packToGo: !item.packToGo }
              : item
          ),
        }))
        return {
          orders: { ...state.orders, [tableId]: { ...existing, rounds: newRounds } },
        }
      }),
    ```
