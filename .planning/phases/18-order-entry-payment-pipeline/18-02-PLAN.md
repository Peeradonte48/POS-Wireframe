**Imports to add:**
```typescript
import { useMemo } from 'react'
import { useOrderStore } from '@/stores/order.store'
import { MENU_ITEMS } from '@/lib/mock-data/menu'
```

**Inside the `TakeawayCard` component body**, add the derivation before the return:
```typescript
const orderData = useOrderStore((s) => s.orders[order.orderId])
const itemsSummary = useMemo(() => {
  if (!orderData || order.status === 'Taking') return 'No items yet'
  const items = orderData.rounds
    .flatMap((r) => r.items)
    .filter((i) => i.status !== 'voided')
  if (items.length === 0) return 'No items yet'
  const grouped = items.reduce<Record<string, number>>((acc, item) => {
    acc[item.menuItemId] = (acc[item.menuItemId] ?? 0) + item.quantity
    return acc
  }, {})
  const parts = Object.entries(grouped).map(([id, qty]) => {
    const label = MENU_ITEMS.find((m) => m.id === id)?.name ?? id
    return `${qty}x ${label}`
  })
  const MAX_ITEMS = 3
  if (parts.length <= MAX_ITEMS) return parts.join(', ')
  return `${parts.slice(0, MAX_ITEMS).join(', ')} +${parts.length - MAX_ITEMS} more`
}, [orderData, order.status])
```

**In the JSX**, replace the static `{order.itemsSummary}` with the derived `{itemsSummary}`:
```tsx
<span className="text-xs text-muted-foreground mt-0.5">{itemsSummary}</span>
```

Do not change anything else in the component — the status badge, CTAs, and cancel dialog remain unchanged.

Note on Zustand selector safety (per CLAUDE.md): `useOrderStore((s) => s.orders[order.orderId])` selects the raw order object — this is safe because `orders[id]` returns the same object reference when no mutation has occurred. `useMemo` then derives the string from this stable reference.
