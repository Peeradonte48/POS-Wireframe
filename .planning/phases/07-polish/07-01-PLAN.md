In globals.css:

1. Replace `--color-brand-primary: oklch(0.55 0.18 262)` in @theme block with:
```css
--color-brand-red:        oklch(0.52 0.22 27);
--color-brand-red-hover:  oklch(0.46 0.22 27);
--color-brand-red-muted:  oklch(0.52 0.22 27 / 15%);
```
Keep `--animate-shake` and the shake @keyframes unchanged.

2. Fix @custom-variant dark selector — change from:
`@custom-variant dark (&:is(.dark *));`
to:
`@custom-variant dark (&:where(.dark, .dark *));`
This is CRITICAL: next-themes applies .dark to `<html>` not a parent element. The `is(.dark *)` selector doesn't match the html element itself; `where(.dark, .dark *)` matches both.

3. In `:root`, override --primary with brand red:
```css
--primary:            oklch(0.52 0.22 27);
--primary-foreground: oklch(0.99 0 0);
```

4. In `.dark`, override --primary with dark-mode brand red (lighter for contrast on dark surfaces):
```css
--primary:            oklch(0.63 0.22 27);
--primary-foreground: oklch(0.10 0 0);
```

5. In @theme inline block, add after existing --font-sans line:
```css
--font-sans: var(--font-inter), var(--font-noto-jp), system-ui, sans-serif;
```
(This overrides the current `--font-sans: var(--font-sans)` self-reference.)

import { useTheme } from 'next-themes'
import { Button } from '@/components/ui/button'

export function ThemeToggle() {
  const { theme, setTheme } = useTheme()
  return (
    <Button
      variant="ghost"
      size="icon"
      className="h-9 w-9"
      onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
      aria-label="Toggle theme"
    >
      <span className="text-sm select-none dark:hidden">☀</span>
      <span className="text-sm select-none hidden dark:inline">☾</span>
    </Button>
  )
}

    <html lang="th" suppressHydrationWarning>
      <body
