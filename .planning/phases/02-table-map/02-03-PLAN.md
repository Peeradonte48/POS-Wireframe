**File 2: `src/app/(app)/table-map/page.tsx` — final wiring**

Replace the Plan 02 placeholder page with the fully wired version. Manages three pieces of state:
1. `selectedTable: TableRecord | null` — which table the bottom sheet shows
2. `openModalTableId: string | null` — which table's Open Table modal is open

```typescript
'use client'
import { useState } from 'react'
import { TableGrid } from '@/components/table-map/TableGrid'
import { TableBottomSheet } from '@/components/table-map/TableBottomSheet'
import { OpenTableModal } from '@/components/table-map/OpenTableModal'
import type { TableRecord } from '@/stores/table.store'

export default function TableMapPage() {
  const [selectedTable, setSelectedTable] = useState<TableRecord | null>(null)
  const [openModalTableId, setOpenModalTableId] = useState<string | null>(null)

  const handleTableTap = (table: TableRecord) => {
    setSelectedTable(table)
  }

  const handleCloseSheet = () => {
    setSelectedTable(null)
  }

  const handleOpenTableModal = () => {
    if (selectedTable) {
      setOpenModalTableId(selectedTable.id)
      setSelectedTable(null)   // close bottom sheet while modal is open
    }
  }

  const handleCloseModal = () => {
    setOpenModalTableId(null)
  }

  return (
    <div className="min-h-full">
      <TableGrid onTableTap={handleTableTap} />
      <TableBottomSheet
        table={selectedTable}
        onClose={handleCloseSheet}
        onOpenTableModal={handleOpenTableModal}
      />
      <OpenTableModal
        tableId={openModalTableId}
        onClose={handleCloseModal}
      />
    </div>
  )
}
```

Note: `TableBottomSheet` reads live data from `useTableStore` internally, so even though the `table` prop is a snapshot, the sheet's display of waiterName/note will reflect store state via local state sync.

After calling `openTable` in the modal and closing it, the tile on the grid auto-updates because `TableGrid` reads from `useTableStore` reactively.
